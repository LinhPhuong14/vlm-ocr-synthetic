# genalog renderer. `genalog` itself is deliberately NOT listed: its source is
# vendored in this directory, and `render.py` lives here too, so `generators/
# genalog/` is sys.path[0] whenever it runs and the vendored tree wins over any
# installed copy. Installing the package as well would leave two of them with
# one silently shadowing the other.
#
# That vendoring also side-steps genalog's pins -- numpy 1.18.1, WeasyPrint 51,
# scikit-image 0.16.2, none of which has a wheel for Python 3.9+ -- because
# nothing on the path this repository calls touches those APIs.

                          # render.py rasterises the PDF instead
pymupdf>=1.23             # PDF -> raster
jinja2>=3                 # genalog's template layer
numpy>=1.24
opencv-python-headless>=4.8,<5
pillow>=10
PyYAML>=6
